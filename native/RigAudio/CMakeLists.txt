cmake_minimum_required(VERSION 3.22)
project(AliveRigAudio LANGUAGES C CXX OBJCXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_OSX_DEPLOYMENT_TARGET 14.0)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 2cdfca8feb300fb424002ba2c2751569e5bacb64)
FetchContent_Declare(NAM GIT_REPOSITORY https://github.com/sdatkinson/NeuralAmpModelerCore.git GIT_TAG 1f42f88535884450104b8711d7595019afa0495b GIT_SUBMODULES Dependencies/eigen)
FetchContent_MakeAvailable(JUCE)
FetchContent_GetProperties(NAM)
if(NOT nam_POPULATED)
    FetchContent_Populate(NAM)
endif()
file(GLOB NAM_SOURCES "${nam_SOURCE_DIR}/NAM/*.cpp" "${nam_SOURCE_DIR}/NAM/*/*.cpp")
# Compile registrars directly into the dylib, so no static dead-stripping can lose A2.
add_library(AliveRigAudio SHARED RigAudio.cpp ${NAM_SOURCES}
    dsp/DcBlocker.cpp dsp/NoiseGate.cpp dsp/SpringReverb.cpp dsp/SafetyLimiter.cpp dsp/RigEffects.cpp)
target_include_directories(AliveRigAudio PRIVATE . "${nam_SOURCE_DIR}" "${nam_SOURCE_DIR}/Dependencies/eigen" "${nam_SOURCE_DIR}/Dependencies/nlohmann")
target_compile_definitions(AliveRigAudio PRIVATE NAM_SAMPLE_FLOAT NAM_ENABLE_A2_FAST JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(AliveRigAudio PRIVATE juce::juce_audio_devices juce::juce_dsp)
set_target_properties(AliveRigAudio PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR TRUE)
add_executable(rig_audio_test RigAudioTest.cpp)
target_link_libraries(rig_audio_test PRIVATE AliveRigAudio)
enable_testing()
add_test(NAME rig_audio COMMAND rig_audio_test)
add_executable(rig_effects_test RigEffectsTest.cpp dsp/RigEffects.cpp)
target_link_libraries(rig_effects_test PRIVATE juce::juce_dsp)
target_compile_definitions(rig_effects_test PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
add_test(NAME rig_effects COMMAND rig_effects_test)
